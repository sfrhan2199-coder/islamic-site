const express = require("express");
const session = require("express-session");
const bcrypt = require("bcrypt");
const sqlite3 = require("sqlite3").verbose();
const multer = require("multer");

const app = express();
const db = new sqlite3.Database("database.sqlite");
const upload = multer({ dest: "uploads/" });

app.use(express.urlencoded({ extended: true }));
app.use(express.static("."));
app.use(session({
  secret: "arabicSiteSecret",
  resave: false,
  saveUninitialized: true
}));

/* ================== قاعدة البيانات ================== */
db.serialize(() => {
  db.run(`
    CREATE TABLE IF NOT EXISTS users (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      username TEXT UNIQUE,
      email TEXT UNIQUE,
      password TEXT,
      role TEXT,
      reset_code TEXT
    )
  `);

  db.run(`
    CREATE TABLE IF NOT EXISTS contents (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      title TEXT,
      body TEXT,
      category TEXT,
      image TEXT,
      visible INTEGER DEFAULT 1
    )
  `);
});

function كلمة_مرور_قوية(p) {
  return p.length >= 8 && /\d/.test(p) && /[A-Za-z]/.test(p);
}

/* ================== تسجيل ================== */
app.get("/register", (req, res) => {
  res.send(صفحة(`
    <form class="box" method="post">
      <h3>إنشاء حساب جديد</h3>
      <input name="username" placeholder="اسم المستخدم" required>
      <input name="email" placeholder="البريد الإلكتروني" required>
      <input type="password" name="password" placeholder="كلمة المرور" required>
      <button>تسجيل</button>
    </form>
  `));
});

app.post("/register", async (req, res) => {
  if (!كلمة_مرور_قوية(req.body.password))
    return res.send("❌ كلمة المرور ضعيفة (8 أحرف + رقم + حرف)");

  const hash = await bcrypt.hash(req.body.password, 10);
  db.run(
    "INSERT INTO users (username,email,password,role) VALUES (?,?,?,?)",
    [req.body.username, req.body.email, hash, "user"],
    () => res.redirect("/login")
  );
});

/* ================== دخول ================== */
app.get("/login", (req, res) => {
  res.send(صفحة(`
    <form class="box" method="post">
      <h3>تسجيل الدخول</h3>
      <input name="username" placeholder="اسم المستخدم">
      <input type="password" name="password" placeholder="كلمة المرور">
      <button>دخول</button>
      <a href="/forgot">نسيت كلمة المرور؟</a>
    </form>
  `));
});

app.post("/login", (req, res) => {
  db.get("SELECT * FROM users WHERE username=?", [req.body.username], async (e, u) => {
    if (u && await bcrypt.compare(req.body.password, u.password)) {
      req.session.user = u;
      res.redirect("/");
    } else {
      res.send("❌ بيانات الدخول غير صحيحة");
    }
  });
});

/* ================== نسيت كلمة المرور ================== */
app.get("/forgot", (req, res) => {
  res.send(صفحة(`
    <form class="box" method="post">
      <h3>استرجاع كلمة المرور</h3>
      <input name="email" placeholder="البريد الإلكتروني">
      <button>إرسال الرمز</button>
    </form>
  `));
});

app.post("/forgot", (req, res) => {
  const code = Math.floor(100000 + Math.random() * 900000).toString();
  db.run(
    "UPDATE users SET reset_code=? WHERE email=?",
    [code, req.body.email],
    () => res.send(`رمز الاسترجاع: <b>${code}</b> <br><a href="/reset">تغيير كلمة المرور</a>`)
  );
});

app.get("/reset", (req, res) => {
  res.send(صفحة(`
    <form class="box" method="post">
      <input name="code" placeholder="رمز الاسترجاع">
      <input name="password" placeholder="كلمة مرور جديدة">
      <button>تغيير</button>
    </form>
  `));
});

app.post("/reset", async (req, res) => {
  if (!كلمة_مرور_قوية(req.body.password))
    return res.send("❌ كلمة المرور ضعيفة");

  const hash = await bcrypt.hash(req.body.password, 10);
  db.run(
    "UPDATE users SET password=?, reset_code=NULL WHERE reset_code=?",
    [hash, req.body.code],
    () => res.redirect("/login")
  );
});
